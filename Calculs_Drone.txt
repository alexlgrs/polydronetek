clear; clc;

%% === PARAMETERS ===
syms phi dphi theta dtheta psi dpsi x dx y dy z dz
m = 0; %[kg] Masse total du quadrirotor
g = 9.80665; %[m/s^2] Gravité
Jr = 0; % Inertie des rotors
Kd = 0; % Coef de traînée
Kp = 0; % Coef de portance
Kfa = 0; % Coefs frottements aérodynamiques
Kft = 0; % Coefs de force de traînée de translation
d = 0; % Distance entre le centre de masse du quadrirotor et l’axe de rotation du rotor

% Vitesse de rotation des moteurs
w1 = 0;
w2 = 0;
w3 = 0;
w4 = 0;

% Inertie
Ix = 0;
Iy = 0;
Iz = 0;
% Coefs de force de traînée de translation
Ktx = 0;
Kty = 0;
Ktz = 0;
% Coefs frottements aérodynamiques
Kax = 0;
Kay = 0;
Kaz = 0;


% Forces de poussée/portance
F1 = Kp*w1*w1;
F2 = Kp*w2*w2;
F3 = Kp*w3*w3;
F4 = Kp*w4*w4;

omega_r = w1 - w2 + w3 - w4;
u1 = F1 + F2 + F3 + F4;              % Poussée des 4 rotors
u2 = F3 - F1;                        % Différence de poussée rotor gauche et droit
u3 = F4 - F2;                        % Différence de poussée rotor avant et arrière
u4 = Kd*(w1^2 - w2^2 + w3^2 - w4^2); % Résultante des couples des 4 rotors
ux = cos(phi)*sin(theta)*cos(psi) + sin(phi)*sin(psi);
uy = cos(phi)*sin(theta)*sin(psi) - sin(phi)*cos(psi);

% Lumped Constants
a1 = (Iy-Iz)/Ix;
a2 = -Kax/Ix;
a3 = -Jr/Ix;
a4 = (Iz-Ix)/Iy;
a5 = -Kay/Iy;
a6 = -Jr/Iy;
a7 = (Ix-Iy)/Iz;
a8 = -Kaz/Iz;
a9 = -Ktx/m;
a10 = -Kty/m;
a11 = -Ktz/m;
b1 = d/Ix;
b2 = d/Iy;
b3 = 1/Iz;



%% === SYSTEM EQUATIONS ===
% Equations of the system
ddphi = a1*dtheta*dpsi + a2*dphi*dphi + a3*omega_r*dtheta + b1*u2;
ddtheta = a4*dphi*dpsi + a5*dtheta*dtheta + a6*omega_r*dphi + b2*u3;
ddpsi = a7*dphi*dtheta + a8*dpsi*dpsi + b3*u4;
ddx = a9*dx + (ux/m)*u1;
ddy = a10*dy + (uy/m)*u1;
ddz = a11*dz - g + (cos(phi)*cos(theta)/m)*u1;

f = [dphi; ddphi; dtheta; ddtheta; dpsi; ddpsi; dx; ddx; dy; ddy; dz; ddz];
